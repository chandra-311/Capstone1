📦 INSTALLATION (One-time setup)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Navigate to project directory:
   $ cd /home/ubuntu/Desktop/LoanApplication

2. Create environment file:
   $ cp .env.example .env

3. Add your API key to .env:
   $ nano .env
   # Edit: ANTHROPIC_API_KEY=your-key-here
   # Get key from: https://console.anthropic.com/api_keys

4. Install dependencies:
   $ pip install -r requirements.txt

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🚀 RUN ALL SERVICES (Automated)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

$ ./start_all_services.sh

This automatically starts all 6 services:
  ✓ ApplicantDB MCP Server     (port 8001)
  ✓ RiskRulesDB MCP Server     (port 8002)
  ✓ DecisionSynthesis MCP      (port 8003)
  ✓ NotificationSystem MCP     (port 8004)
  ✓ FastAPI Microservice       (port 8000)
  ✓ Streamlit UI               (port 8501)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🌐 ACCESS THE SYSTEM
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Open Streamlit UI:
   👉 http://localhost:8501

2. Fill in loan application form:
   - Applicant ID (e.g., APPL-001)
   - Age, Income, Employment Type
   - Credit Score, Loan Amount, Loan Tenure
   - Existing Liabilities, Location

3. Click "🚀 Submit Application"

4. View Decision, Risk Score, and Analysis

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📊 ALTERNATIVE ACCESS METHODS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

API Documentation:
   👉 http://localhost:8000/docs

Submit application via curl:
   $ curl -X POST http://localhost:8000/api/apply \
     -H "Content-Type: application/json" \
     -d '{
       "applicant_id": "APPL-001",
       "age": 35,
       "income": 75000,
       "employment_type": "full_time",
       "credit_score": 720,
       "loan_amount": 250000,
       "loan_tenure_months": 60,
       "existing_liabilities": 50000,
       "location": "California"
     }'

Check application status:
   $ curl http://localhost:8000/api/status/APPL-001

🛑 STOP SERVICES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

If using automated script:
   - Press Ctrl+C in the terminal where start_all_services.sh is running
   - All services will automatically shut down

If running manually:
   - Press Ctrl+C in each terminal window

🆘 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Port already in use?
   $ lsof -ti:8000 | xargs kill -9  # Kill process on port 8000

Module not found?
   $ pip install -r requirements.txt --force-reinstall

API key error?
   $ cat .env | grep ANTHROPIC_API_KEY
   # Get key from: https://console.anthropic.com/api_keys

Services won't connect?
   $ netstat -tuln | grep -E '800[0-4]'  # Check ports are listening

Streamlit not loading?
   $ streamlit run ui/streamlit_app.py --logger.level=debug
